cmake_minimum_required(VERSION 3.20)
project(ParadeAnimation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)

add_executable(timeline_test code/tests/timeline_test.cpp)
target_include_directories(timeline_test PRIVATE code/include)
if(MSVC)
    target_compile_options(timeline_test PRIVATE /UNDEBUG)
else()
    target_compile_options(timeline_test PRIVATE -UNDEBUG)
endif()
add_test(NAME timeline COMMAND timeline_test)

option(BUILD_SCENE "Build the OpenGL scene (requires dependency downloads)" ON)
if(BUILD_SCENE)
    include(FetchContent)
    if(NOT CMAKE_POLICY_VERSION_MINIMUM)
        set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
    endif()
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(ONLY_LIBS ON CACHE BOOL "" FORCE)
    set(glew-cmake_BUILD_SHARED OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
    FetchContent_Declare(glew GIT_REPOSITORY https://github.com/Perlmint/glew-cmake.git GIT_TAG glew-cmake-2.2.0 GIT_SHALLOW TRUE)
    FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.1 GIT_SHALLOW TRUE)
    FetchContent_Declare(tinyobjloader GIT_REPOSITORY https://github.com/tinyobjloader/tinyobjloader.git GIT_TAG v2.0.0rc13 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(glfw glew glm tinyobjloader)
    # Gate walls contain concave polygons around the archways. Use robust
    # triangulation so these polygons retain their complete surface.
    target_compile_definitions(tinyobjloader PRIVATE TINYOBJLOADER_USE_MAPBOX_EARCUT)
    if(BUILD_TESTING)
        add_executable(wall_triangulation_test code/tests/wall_triangulation_test.cpp)
        target_link_libraries(wall_triangulation_test PRIVATE tinyobjloader)
        target_compile_definitions(wall_triangulation_test PRIVATE
            WALL_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/code/tests/data/gate_wall.obj.txt")
        add_test(NAME wall_triangulation COMMAND wall_triangulation_test)
    endif()
    FetchContent_Declare(stb GIT_REPOSITORY https://github.com/nothings/stb.git GIT_TAG f0569113c93ad095470c54bf34a17b36646bbbb5 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(stb)
    if(APPLE)
        foreach(_glew_target libglew_static)
            get_target_property(_glew_libs ${_glew_target} INTERFACE_LINK_LIBRARIES)
            if(_glew_libs)
                list(FILTER _glew_libs EXCLUDE REGEX "AGL")
                set_target_properties(${_glew_target} PROPERTIES INTERFACE_LINK_LIBRARIES "${_glew_libs}")
            endif()
        endforeach()
    endif()
    find_package(OpenGL REQUIRED)
    add_executable(parade_scene code/src/main.cpp code/src/Shader.cpp code/src/Mesh.cpp)
    target_include_directories(parade_scene PRIVATE code/include)
    target_include_directories(parade_scene SYSTEM PRIVATE ${stb_SOURCE_DIR})
    target_link_libraries(parade_scene PRIVATE glfw libglew_static glm::glm tinyobjloader OpenGL::GL)
    target_compile_definitions(parade_scene PRIVATE GLEW_STATIC PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
    if(MSVC)
        target_compile_options(parade_scene PRIVATE /W4 /utf-8)
    else()
        target_compile_options(parade_scene PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
